\chapter*{Chapter 1 glossary}\addcontentsline{toc}{chapter}{Chapter 1 glossary}
\begin{description}
\item[Walk] A vertex sequence respecting directed edges; repeats are allowed. See Section~\ref{sec:graph}.
\item[Hamiltonian path] A simple path containing every graph vertex exactly once. See Section~\ref{sec:graph}.
\item[Decision problem] A problem whose required answer is yes or no. See Section~\ref{sec:graph}.
\item[Domain] A contiguous sequence assigned a recognition role in a molecular design. See Section~\ref{sec:encoding}.
\item[Reverse complement] The complementary DNA sequence rewritten in the same 5-prime to 3-prime convention. See Section~\ref{sec:encoding}.
\item[Multiset] A collection retaining how many copies of each object occur. See Section~\ref{sec:population}.
\item[Support] The distinct objects present with nonzero multiplicity. See Section~\ref{sec:population}.
\item[Hybridization] Association through complementary sequence interactions. See Section~\ref{sec:readout}.
\item[Ligation] Covalent connection between suitable adjacent backbone ends. See Section~\ref{sec:readout}.
\item[PCR] Primer-directed cyclic DNA amplification. See Section~\ref{sec:readout}.
\item[Affinity selection] Retention supported by a probe recognizing a target subsequence. See Section~\ref{sec:readout}.
\item[Certificate] A proposed witness whose correctness can be checked against the instance. See Section~\ref{sec:complexity}.
\item[Sampling failure] Missing a target event in the declared number of draws. See Section~\ref{sec:scaling}.
\end{description}
