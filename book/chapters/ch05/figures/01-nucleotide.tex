% 2'-deoxyadenosine 5'-monophosphate (dAMP) as a skeletal structure.
% Ring atoms are placed on regular polygons with bond length \b; the purine is
% a hexagon fused to a pentagon at C4-C5; the furanose is a pentagon with O4' up.
\begin{tikzpicture}[x=1cm,y=1cm,
  atm/.style={font=\sffamily\small,text=Ink,inner sep=1.2pt,fill=white},
  hat/.style={font=\sffamily\small,text=Ink,inner sep=1.2pt},
  bd/.style={line width=1.05pt,Ink},
  db/.style={line width=1.05pt,Ink,double,double distance=1.7pt},
  num/.style={font=\sffamily\scriptsize\bfseries,inner sep=.5pt}]

\begin{scope}[on background layer]
 \fill[BaseA!13,rounded corners=10pt] (-2.75*\b,-1.25*\b) rectangle (1.45*\b,2.3*\b);
 \fill[DNAblue!9,rounded corners=10pt] (-4.6*\b,-3.95*\b) rectangle (-1.65*\b,-.55*\b);
 \fill[StrandAmber!14,rounded corners=10pt] (-6.75*\b,-1.1*\b) rectangle (-4.45*\b,2.05*\b);
\end{scope}
% ---------------------------------------------------------------- adenine

\draw[db] (C4)--(C5); \draw[bd] (C5)--(C6); \draw[bd] (C5)--(N7); \draw[db] (N7)--(C8);
\draw[bd] (C8)--(N9); \draw[bd] (N9)--(C4); \draw[bd] (C6)--(N6);
\foreach \p/\t in {N1/1,N7/7,N9/9}{\node[num,text=BaseA!70!black,anchor=south west] at ($(\p)+(.06,.05)$) {\t};}
